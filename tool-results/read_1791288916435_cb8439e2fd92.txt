   123→   123→function findService(text: string, services: Svc[]): Svc | null {
   124→   124→  const t = normalize(text);
   125→   125→  if (!t) return null;
   126→   126→  // exact-ish match first
   127→   127→  for (const s of services) {
   128→   128→    if (normalize(s.service_name) === t || normalize(s.service_id) === t) return s;
   129→   129→  }
   130→   130→  // substring match
   131→   131→  for (const s of services) {
   132→   132→    const n = normalize(s.service_name);
   133→   133→    if (n.length >= 4 && (t.includes(n) || n.includes(t))) return s;
   134→   134→  }
   135→   135→  // category match ("pan card ka kaam", "income certificate lagana he")
   136→   136→  for (const s of services) {
   137→   137→    const words = normalize(s.service_name).split(/(?=[A-Z])/);
   138→   138→    if (words.length && words[0].length >= 4 && t.includes(words[0])) return s;
   139→   139→  }
   140→   140→  return null;
   141→   141→}

   176→   176→async function askCurrentField(state: WebState): Promise<BotReply | null> {
   177→   177→  const fields = state.fields ?? [];

   179→   179→  if (i >= fields.length) return null;
   180→   180→  const f = fields[i];

   182→   182→  if (!f.required) chips.push("SKIP");

   184→   184→    return {
   185→   185→      text: `${f.question}\n\n${f.options.map((o, k) => `${k + 1}. ${o}`).join("\n")}`,
   186→   186→      chips: [...f.options.slice(0, 4), ...chips],

   362→   362→export async function processInput(
   363→   363→  session: EngineSession,
   364→   364→  input: EngineInput
   365→   365→): Promise<EngineResult> {
   366→   366→  const text = (input.text ?? "").trim();
   367→   367→  const att = input.attachment;
   368→   368→  let stage = session.stage;
   369→   369→  let state: WebState = session.state ?? resetState();
   370→   370→  const replies: BotReply[] = [];
   371→   371→  let patchOut: Partial<EngineSession> = {};
   372→   372→
   373→   373→  const setStage = (s: string) => (stage = s);
   374→   374→  const say = (r: BotReply) => replies.push(r);
   375→   375→  const phoneForAI = session.phone || `web_${hashId(session.sessionId)}`;
   376→   376→
   377→   377→  // ---------- GLOBAL CANCEL (any stage) ----------
   378→   378→  if (text && isCancel(text) && stage !== STAGE.NAME && stage !== STAGE.PHONE) {
   379→   379→    setStage(STAGE.MENU);
   380→   380→    state = resetState();
   381→   381→    say({ text: "Theek he, cancel kar diya. Koi baat nahi! 😊" });
   382→   382→    say(await menuReply());
   383→   383→    return { replies, patch: { stage, state } };
   384→   384→  }
   385→   385→
   386→   386→  switch (stage) {
   387→   387→    // ================= STEP 1: NAME =================
   388→   388→    case STAGE.NAME: {
   389→   389→      const looksPhone = /^\+?\d[\d\s-]{7,}$/.test(text);
   390→   390→      const isGreeting = /^(hi|hii|hiii|hello|namaste|namaskar|hey|salam|salaam|assalam|good (morning|afternoon|evening)|start|shuru)\b/i.test(text);
   391→   391→      if (isGreeting) {
   392→   392→        say({
   393→   393→          text:
   394→   394→            "Namaste! 🙏 Main *Ravi* hoon — CSC Smart Seva ka assistant.\n\nSarkari kaam ghar baithe — PAN card, ITR, GST, certificate sab yahin se.\n\nChaliye shuru karte hein — *aapka pura naam* likhein.",
   395→   395→        });
   396→   396→        break;
   397→   397→      }
   398→   398→      if (!text || text.length < 2 || looksPhone || !/[a-zA-Z]/.test(text)) {
   399→   399→        say({
   400→   400→          text: looksPhone
   401→   401→            ? "Pehle aapka naam chahiye 😊 Naam likhein, phir number."
   402→   402→            : "Naam thoda sahi se likhein (jaise: Ramesh Kumar).",
   403→   403→        });
   404→   404→        break;
   405→   405→      }
   406→   406→      const name = text.slice(0, 60).replace(/\s+/g, " ");
   407→   407→      session.name = name;
   408→   408→      patchOut.name = name;
   409→   409→      setStage(STAGE.PHONE);
   410→   410→      say({
   411→   411→        text: `Namaste *${name}*! 🙏\nMain Ravi hoon — CSC Smart Seva ka assistant.\n\nAb aapka *mobile number* likhein (10 digit) — isi se aapki customer ID banegi.`,
   412→   412→      });
   413→   413→      break;
   414→   414→    }
   415→   415→
   416→   416→    // ================= STEP 2: PHONE + ID =================
   417→   417→    case STAGE.PHONE: {
   418→   418→      const v = validateField("MOBILE", text);
   419→   419→      if (!v.ok) {
   420→   420→        say({ text: `${v.reason} Dobara likhein.` });
   421→   421→        break;
   422→   422→      }
   423→   423→      const phone = v.value;
   424→   424→      session.phone = phone;
   425→   425→      const cid = `CUST-2026-${rand(5)}`;
   426→   426→      session.customerId = cid;
   427→   427→      state = resetState();
   428→   428→
   429→   429→      try {
   430→   430→        await db.customer.upsert({
   431→   431→          where: { phone },
   432→   432→          update: { name: session.name },
   433→   433→          create: { phone, name: session.name },
   434→   434→        });
   435→   435→      } catch { /* non-fatal */ }
   436→   436→      try {
   437→   437→        upsertCustomerN8n(phone, session.name);
   438→   438→      } catch { /* n8n down */ }
   439→   439→
   440→   440→      setStage(STAGE.MENU);
   441→   441→      say({
   442→   442→        text:
   443→   443→          `🎉 *Aapki ID ban gayi!*\n\n` +
   444→   444→          `👤 Customer ID: *${cid}*\n` +
   445→   445→          `📱 Mobile: ${phone}\n` +
   446→   446→          `🧑 Naam: ${session.name}\n\n` +
   447→   447→          `Ab se har seva isi ID pe hogi. services dikhata hoon 👇`,
   448→   448→      });
   449→   449→      say(await menuReply());
   450→   450→      break;
   451→   451→    }
   452→   452→
   453→   453→    // ================= MENU =================
   454→   454→    case STAGE.MENU: {
   455→   455→      const services = await getServices();
   456→   456→      const svc = text ? findService(text, services) : null;
   457→   457→      if (svc) {
   458→   458→        setStage(STAGE.OFFER);
   459→   459→        say(...(await startService(state, svc)));
   460→   460→        break;
   461→   461→      }
   462→   462→      // nayi seva chip / general question → AI
   463→   463→      if (text) {
   464→   464→        const ai = await aiGeneralChat(phoneForAI, text);
   465→   465→        if (ai) {
   466→   466→          say({ text: ai });
   467→   467→          say({ text: "Koi seva chahiye to card par dabayein 👇", cards: services });
   468→   468→        } else {
   469→   469→          say({ text: "Main services ke bare me batata hoon 👇" });
   470→   470→          say(await menuReply());
   471→   471→        }
   472→   472→        break;
   473→   473→      }
   474→   474→      say(await menuReply());
   475→   475→      break;
   476→   476→    }
   477→   477→
   478→   478→    // ================= OFFER (fee confirm) =================
   479→   479→    case STAGE.OFFER: {
   480→   480→      const services = await getServices();
   481→   481→      const svc =
   482→   482→        services.find((s) => s.service_id === state.serviceId) ??
   483→   483→        (text ? findService(text, services) : null);
   484→   484→      if (!svc) {
   485→   485→        setStage(STAGE.MENU);
   486→   486→        state = resetState();
   487→   487→        say({ text: "Service select karein 👇" });
   488→   488→        say(await menuReply());
   489→   489→        break;
   490→   490→      }
   491→   491→      if (isConfirm(text)) {
   492→   492→        const next = await askCurrentField(state);
   493→   493→        if (next) {
   494→   494→          setStage(STAGE.FIELDS);
   495→   495→          say({ text: `Badhiya! Kuch details chahiye — sirf ${state.fields?.length ?? 0} chhote sawal 👇` });
   496→   496→          say(next);
   497→   497→        } else {
   498→   498→          const d = await askCurrentDoc(state);
   499→   499→          if (d) {
   500→   500→            setStage(STAGE.DOCS);
   501→   501→            say({ text: `Ab documents chahiye 📄` });
   502→   502→            say(d);
   503→   503→          } else {
   504→   504→            setStage(STAGE.PAYMENT);
   505→   505→            say(await beginPayment(svc));
   506→   506→          }
   507→   507→        }
   508→   508→        break;
   509→   509→      }
   510→   510→      // side question / different service
   511→   511→      const otherSvc = findService(text, services);
   512→   512→      if (otherSvc && otherSvc.service_id !== state.serviceId) {
   513→   513→        say(...(await startService(state, otherSvc)));
   514→   514→        break;
   515→   515→      }
   516→   516→      if (isMenu(text)) {
   517→   517→        setStage(STAGE.MENU);
   518→   518→        state = resetState();
   519→   519→        say(await menuReply());
   520→   520→        break;
   521→   521→      }
   522→   522→      if (text) {
   523→   523→        const ai = await aiGeneralChat(phoneForAI, text);
   524→   524→        if (ai) say({ text: ai });
   525→   525→      }
   526→   526→      say({
   527→   527→        text: `${svc.service_name} ke liye aage badhna he to *Confirm karo* dabayein 👇`,
   528→   528→        chips: ["✅ Confirm karo", "📋 Menu wapas"],
   529→   529→      });
   530→   530→      break;
   531→   531→    }
   532→   532→
   533→   533→    // ================= FIELDS =================
   534→   534→    case STAGE.FIELDS: {
   535→   535→      const fields = state.fields ?? [];
   536→   536→      let i = state.fidx ?? 0;
   537→   537→      if (i >= fields.length) {
   538→   538→        // shouldn't happen; jump to docs/payment
   539→   539→        const d = await askCurrentDoc(state);
   540→   540→        if (d) {
   541→   541→          setStage(STAGE.DOCS);
   542→   542→          say(d);
   543→   543→        } else {
   544→   544→          setStage(STAGE.PAYMENT);
   545→   545→          const svc = (await getServices()).find((s) => s.service_id === state.serviceId);
   546→   546→          if (svc) say(await beginPayment(svc));
   547→   547→        }
   548→   548→        break;
   549→   549→      }
   550→   550→      const f = fields[i];
   551→   551→
   552→   552→      // user sent a file while a text answer was expected
   553→   553→      if (att) {
   554→   554→        say({
   555→   555→          text: `Ye sawal likhkar jawab dena he: ${f.question}`,
   556→   556→        });
   557→   557→        break;
   558→   558→      }
   559→   559→
   560→   560→      if (isMenu(text)) {
   561→   561→        setStage(STAGE.MENU);
   562→   562→        state = resetState();
   563→   563→        say(await menuReply());
   564→   564→        break;
   565→   565→      }
   566→   566→
   567→   567→      const canSkip = !f.required || (state.fails ?? 0) >= 3;
   568→   568→      if (isSkip(text)) {
   569→   569→        if (canSkip) {
   570→   570→          state.collected = { ...(state.collected ?? {}), [f.field_key]: "SKIP" };
   571→   571→          state.fails = 0;
   572→   572→          i += 1;
   573→   573→          state.fidx = i;
   574→   574→          const nxt = await askCurrentField(state);
   575→   575→          if (nxt) {
   576→   576→            say({ text: `${ack()} — SKIP 👍` });
   577→   577→            say(nxt);
   578→   578→          } else {
   579→   579→            const d = await askCurrentDoc(state);
   580→   580→            if (d) {
   581→   581→              setStage(STAGE.DOCS);
   582→   582→              say({ text: "Sawal complete! Ab documents 📄" });
   583→   583→              say(d);
   584→   584→            } else {
   585→   585→              setStage(STAGE.PAYMENT);
   586→   586→              const svc = (await getServices()).find((s) => s.service_id === state.serviceId);
   587→   587→              if (svc) say(await beginPayment(svc));
   588→   588→            }
   589→   589→          }
   590→   590→        } else {
   591→   591→          say({ text: "Ye sawal zaroori he — jawab likhein. (3 baar try ke baad SKIP ho jayega)" });
   592→   592→        }
   593→   593→        break;
   594→   594→      }
   595→   595→
   596→   596→      // AI side-question gate (same as WhatsApp bot)
   597→   597→      const gate = await aiFieldGate({
   598→   598→        phone: phoneForAI,
   599→   599→        name: session.name,
   600→   600→        text,
   601→   601→        state: "FIELDS",
   602→   602→        serviceName: state.serviceName ?? "",
   603→   603→        field: { label: f.label, question: f.question },
   604→   604→        collected: state.collected ?? {},
   605→   605→      });
   606→   606→      if (gate && gate.action === "reply") {
   607→   607→        say({ text: gate.reply });
   608→   608→        say({ text: `Wapas sawal: ${f.question}` });
   609→   609→        break;
   610→   610→      }
   611→   611→
   612→   612→      // validate
   613→   613→      const v = f.field_type === "CHOICE" ? matchChoice(f.options, text) : validateField(f.field_type, text);
   614→   614→      if (!v.ok) {
   615→   615→        state.fails = (state.fails ?? 0) + 1;
   616→   616→        const extra =
   617→   617→          (state.fails ?? 0) >= 3
   618→   618→            ? "\n\n(Tip: *SKIP* likhkar is sawal ko chhod sakte hein — operator baad me bhar dega)"
   619→   619→            : "";
   620→   620→        say({ text: `⚠️ ${v.reason}${extra}\n\n${f.question}` });
   621→   621→        break;
   622→   622→      }
   623→   623→
   624→   624→      state.collected = { ...(state.collected ?? {}), [f.field_key]: v.value };
   625→   625→      state.fails = 0;
   626→   626→      i += 1;
   627→   627→      state.fidx = i;
   628→   628→      const nxt = await askCurrentField(state);
   629→   629→      if (nxt) {
   630→   630→        say({ text: `${ack()} (${i}/${fields.length})` });
   631→   631→        say(nxt);
   632→   632→      } else {
   633→   633→        const d = await askCurrentDoc(state);
   634→   634→        if (d) {
   635→   635→          setStage(STAGE.DOCS);
   636→   636→          say({ text: `Sawal complete! 🎯 Ab *${state.docs?.length ?? 0}* document chahiye 📄` });
   637→   637→          say(d);
   638→   638→        } else {
   639→   639→          setStage(STAGE.PAYMENT);
   640→   640→          const svc = (await getServices()).find((s) => s.service_id === state.serviceId);
   641→   641→          if (svc) say(await beginPayment(svc));
   642→   642→        }
   643→   643→      }
   644→   644→      break;
   645→   645→    }
   646→   646→
   647→   647→    // ================= DOCS =================
   648→   648→    case STAGE.DOCS: {
   649→   649→      const docs = state.docs ?? [];
   650→   650→      let i = state.didx ?? 0;
   651→   651→      if (i >= docs.length) {
   652→   652→        const svc = (await getServices()).find((s) => s.service_id === state.serviceId);
   653→   653→        setStage(STAGE.PAYMENT);
   654→   654→        if (svc) say(await beginPayment(svc));
   655→   655→        break;
   656→   656→      }
   657→   657→      const d = docs[i];
   658→   658→
   659→   659→      if (att) {
   660→   660→        const okType =
   661→   661→          /^image\//.test(att.type) ||
   662→   662→          att.type === "application/pdf" ||
   663→   663→          /\.(jpe?g|png|webp|pdf|heic)$/i.test(att.name);
   664→   664→        if (!okType) {
   665→   665→          say({ text: "Sirf *photo (JPG/PNG)* ya *PDF* chalta he. Dobara bhejein." });
   666→   666→          break;
   667→   667→        }
   668→   668→        if (att.size > 4 * 1024 * 1024) {
   669→   669→          say({ text: "File 4MB se chhoti honi chahiye. Photo thodi compress karke bhejein." });
   670→   670→          break;
   671→   671→        }
   672→   672→        state.docsCollected = {
   673→   673→          ...(state.docsCollected ?? {}),
   674→   674→          [d.doc_key]: { name: att.name, type: att.type || "image/jpeg", size: att.size },
   675→   675→        };
   676→   676→        i += 1;
   677→   677→        state.didx = i;
   678→   678→        const nxt = await askCurrentDoc(state);
   679→   679→        if (nxt) {
   680→   680→          say({ text: `✅ ${d.label} mil gaya! (${i}/${docs.length})` });
   681→   681→          say(nxt);
   682→   682→        } else {
   683→   683→          const svc = (await getServices()).find((s) => s.service_id === state.serviceId);
   684→   684→          setStage(STAGE.PAYMENT);
   685→   685→          say({ text: `✅ ${d.label} mil gaya! Sab documents complete 🎯` });
   686→   686→          if (svc) say(await beginPayment(svc));
   687→   687→        }
   688→   688→        break;
   689→   689→      }
   690→   690→
   691→   691→      if (isMenu(text)) {
   692→   692→        setStage(STAGE.MENU);
   693→   693→        state = resetState();
   694→   694→        say(await menuReply());
   695→   695→        break;
   696→   696→      }
   697→   697→
   698→   698→      if (isSkip(text)) {
   699→   699→        if (!d.required) {
   700→   700→          i += 1;
   701→   701→          state.didx = i;
   702→   702→          state.docsCollected = { ...(state.docsCollected ?? {}), [d.doc_key]: { name: "SKIP", type: "text/plain", size: 0 } };
   703→   703→          const nxt = await askCurrentDoc(state);
   704→   704→          if (nxt) say(nxt);
   705→   705→          else {
   706→   706→            const svc = (await getServices()).find((s) => s.service_id === state.serviceId);
   707→   707→            setStage(STAGE.PAYMENT);
   708→   708→            if (svc) say(await beginPayment(svc));
   709→   709→          }
   710→   710→        } else {
   711→   711→          say({ text: `${d.label} zaroori he — photo/PDF bhejein. (Operator baad me manga sakta he)` });
   712→   712→        }
   713→   713→        break;
   714→   714→      }
   715→   715→
   716→   716→      // text while file expected → AI gate for side questions
   717→   717→      const gate = await aiFieldGate({
   718→   718→        phone: phoneForAI,
   719→   719→        name: session.name,
   720→   720→        text,
   721→   721→        state: "DOCS",
   722→   722→        serviceName: state.serviceName ?? "",
   723→   723→        field: { label: d.label, question: d.question },
   724→   724→        collected: state.collected ?? {},
   725→   725→      });
   726→   726→      if (gate && gate.action === "reply") {
   727→   727→        say({ text: gate.reply });
   728→   728→        say({ text: `Document: ${d.question}`, expectFile: true });
   729→   729→        break;
   730→   730→      }
   731→   731→      say({
   732→   732→        text: `${d.label} ki *photo ya PDF* attach karke bhejein 📎${d.required ? "" : " (ya *SKIP* likhein)"}`,
   733→   733→        expectFile: true,
   734→   734→        chips: d.required ? [] : ["SKIP"],
   735→   735→      });
   736→   736→      break;
   737→   737→    }
   738→   738→
   739→   739→    // ================= PAYMENT =================
   740→   740→    case STAGE.PAYMENT: {
   741→   741→      const services = await getServices();
   742→   742→      const svc = services.find((s) => s.service_id === state.serviceId);
   743→   743→      if (!svc) {
   744→   744→        setStage(STAGE.MENU);
   745→   745→        state = resetState();
   746→   746→        say(await menuReply());
   747→   747→        break;
   748→   748→      }
   749→   749→      if (isPaid(text) || isConfirm(text)) {
   750→   750→        setStage(STAGE.DONE);
   751→   751→        say(...(await finalize(session)));
   752→   752→        break;
   753→   753→      }
   754→   754→      if (text) {
   755→   755→        const gate = await aiFieldGate({
   756→   756→          phone: phoneForAI,
   757→   757→          name: session.name,
   758→   758→          text,
   759→   759→          state: "PAYMENT",
   760→   760→          serviceName: state.serviceName ?? "",
   761→   761→          field: { label: "Payment", question: "Payment karke Paid likhna he" },
   762→   762→          collected: state.collected ?? {},
   763→   763→        });
   764→   764→        if (gate && gate.action === "reply") {
   765→   765→          say({ text: gate.reply });
   766→   766→          say({ text: "Payment ho jaye to *Paid* likhein 👇", chips: ["✅ Paid ho gaya"] });
   767→   767→          break;
   768→   768→        }
   769→   769→        const ai = await aiGeneralChat(phoneForAI, text);
   770→   770→        if (ai) {
   771→   771→          say({ text: ai });
   772→   772→          say({ text: "Payment ho jaye to *Paid* likhein 👇", chips: ["✅ Paid ho gaya"] });
   773→   773→          break;
   774→   774→        }
   775→   775→      }
   776→   776→      say(await beginPayment(svc));
   777→   777→      break;
   778→   778→    }
   779→   779→
   780→   780→    // ================= DONE =================
   781→   781→    case STAGE.DONE: {
   782→   782→      if (isMenu(text) || /nayi|new/i.test(text)) {
   783→   783→        setStage(STAGE.MENU);
   784→   784→        state = resetState();
   785→   785→        say({ text: "Sure! 👇" });
   786→   786→        say(await menuReply());
   787→   787→        break;
   788→   788→      }
   789→   789→      if (text) {
   790→   790→        const ai = await aiGeneralChat(phoneForAI, text);
   791→   791→        if (ai) {
   792→   792→          say({ text: ai });
   793→   793→          say({ text: "Aur koi seva chahiye? 👇", chips: ["🆕 Nayi seva"] });
   794→   794→          break;
   795→   795→        }
   796→   796→      }
   797→   797→      say({ text: "Nayi seva ke liye *Nayi seva* dabayein 👇", chips: ["🆕 Nayi seva"] });
   798→   798→      break;
   799→   799→    }
   800→   800→
   801→   801→    // ================= fallback =================
   802→   802→    default: {
   803→   803→      setStage(STAGE.MENU);
   804→   804→      say(await menuReply());
   805→   805→    }
   806→   806→  }
   807→   807→
   808→   808→  if (!replies.length) {
   809→   809→    replies.push({ text: "Ek minute… samajh nahi aaya. Dobara try karein ya *Menu* likhein." });
   810→   810→  }
   811→   811→
   812→   812→  return { replies, patch: { stage, ...patchOut, state } };
   813→   813→}
   814→   814→
   815→   815→export { STAGE };
